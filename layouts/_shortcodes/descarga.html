(path.Ext $f) -}} {{- $base := path.Base $f -}} {{- $map := dict "sh" "bash" "py" "python" "yaml" "yaml" "yml" "yaml" "sql" "sql" "conf" "nginx" "html" "html" "txt" "text" "toml" "toml" "json" "json" -}} {{- $lang := .Get 1 | default (index $map $ext) | default "text" -}} {{- if or (eq $base "Dockerfile") (hasPrefix $base "Dockerfile") }}{{ $lang = "dockerfile" }}{{ end -}}
